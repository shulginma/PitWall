cmake_minimum_required(VERSION 3.24)
project(PitWall VERSION 0.4.1 LANGUAGES CXX)

include(CTest)
option(PITWALL_USE_GOOGLETEST "Run the contract suite through GoogleTest" OFF)
option(PITWALL_BUILD_GUI "Build the Windows interactive simulator" OFF)
if(PITWALL_BUILD_GUI AND NOT WIN32)
    message(FATAL_ERROR "The current GUI platform adapter supports Windows only; use a headless preset")
endif()
if(PITWALL_BUILD_GUI)
    enable_language(RC)
endif()

add_library(pitwall_options INTERFACE)
target_compile_features(pitwall_options INTERFACE cxx_std_20)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
    target_compile_options(pitwall_options INTERFACE "$<$<COMPILE_LANGUAGE:CXX>:/W4;/permissive-;/utf-8>")
else()
    target_compile_options(pitwall_options INTERFACE "$<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic>")
endif()

add_subdirectory(libs/domain)
add_subdirectory(libs/simulation)
add_subdirectory(libs/track)
add_subdirectory(apps/simulator)
add_subdirectory(apps/analyzer)
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
